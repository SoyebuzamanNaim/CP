int numofone(int n) {
  int ans = 0;
  for (int i = 0; (1ULL << i) <= n; i++) {
    int vag = 1ULL << (i + 1), arektavag = (n + 1) / vag;
    ans += (arektavag * (vag >> 1ULL));

vector<int> topo_sort(int n, const vector<vector<int>> &adj) {
  vector<int> in_deg(n, 0), order;
  for (int u = 0; u < n; ++u)
    for (int v : adj[u]) in_deg[v]++;
  queue<int> q;
  for (int i = 0; i < n; ++i) if (in_deg[i] == 0) q.push(i);
  while (!q.empty()) {
    int u = q.front(); q.pop(); order.push_back(u);
    for (int v : adj[u]) if (--in_deg[v] == 0) q.push(v);
  }
  return (order.size() == n) ? order : vector<int>();
}

  color.assign(n + 1, -1);
  for (int start = 1; start <= n; ++start) {
    if (color[start] != -1) continue;
    queue<int> q; q.push(start); color[start] = 0;
    while (!q.empty()) {
      int u = q.front(); q.pop();
      for (int v : adj[u]) {
        if (color[v] == -1) {
          color[v] = color[u] ^ 1; q.push(v);
        } else if (color[v] == color[u]) return false; // Odd cycle detected
      }
    }
  }
  return true;
}

void dfs_lca(int u, int p, int d) {
  depth[u] = d; up[u][0] = p;
  for (int i = 1; i < LOGN; ++i)
    up[u][i] = (up[u][i - 1] != -1) ? up[up[u][i - 1]][i - 1] : -1;

int get_lca(int u, int v) {
  if (depth[u] < depth[v]) swap(u, v);
  int diff = depth[u] - depth[v];
  for (int i = 0; i < LOGN; ++i)
    if ((diff >> i) & 1) u = up[u][i];
  if (u == v) return u;
  for (int i = LOGN - 1; i >= 0; --i) {
    if (up[u][i] != up[v][i]) { u = up[u][i]; v = up[v][i]; }
  }
  return up[u][0];
}
int tree_dist(int u, int v) {

}

public:
  DisjointSet(int n) {

    return parent[node] = findUPar(parent[node]);
  }

    if (ulpU == ulpV) return;
    if (size[ulpU] < size[ulpV]) {

    } else {
      parent[ulpV] = ulpU; size[ulpU] += size[ulpV];
    }
  }

};

struct Fenwick {
  int n; vector<int> bit;
  Fenwick(int n) : n(n), bit(n + 1, 0) {}
  void add(int idx, int val) {
    for (; idx <= n; idx += idx & -idx) bit[idx] += val;
  }
  int query(int idx) {
    int sum = 0;
    for (; idx > 0; idx -= idx & -idx) sum += bit[idx];
    return sum;
  }
  int range_query(int l, int r) { return query(r) - query(l - 1); }
  // Average = sum / number_of_elements (count = r - l + 1)
  int avg(int l, int r) { return range_query(l, r) / (r - l + 1); } // integer
  double avg_d(int l, int r) { return (double)range_query(l, r) / (r - l + 1); } // decimal
};

struct SegTree {
  int n; vector<int> t;
  SegTree(int n) : n(n), t(2 * n, 0) {}
  void update(int p, int val) {
    for (t[p += n] = val; p > 1; p >>= 1) t[p >> 1] = t[p] + t[p ^ 1];
  }
  int query(int l, int r) { // sum on [l, r]
    int res = 0;
    for (l += n, r += n + 1; l < r; l >>= 1, r >>= 1) {
      if (l & 1) res += t[l++];
      if (r & 1) res += t[--r];
    }
    return res;
  }
};

struct Tree {
  typedef int T;
  static constexpr T unit = INT_MIN;
  T f(T a, T b) { return max(a, b); } // any associative fn
  vector<T> s; int n;
  Tree(int n = 0, T def = unit) : s(2 * n, def), n(n) {}
  void update(int pos, T val) {
    for (s[pos += n] = val; pos /= 2;)
      s[pos] = f(s[pos * 2], s[pos * 2 + 1]);
  }
  T query(int b, int e) { // query [b, e)
    T ra = unit, rb = unit;
    for (b += n, e += n; b < e; b /= 2, e /= 2) {
      if (b % 2) ra = f(ra, s[b++]);
      if (e % 2) rb = f(s[--e], rb);
    }
    return f(ra, rb);
  }
};

const int inf = 1e9;
struct Node {
  Node *l = 0, *r = 0;
  int lo, hi, mset = inf, madd = 0, val = -inf;
  Node(int lo, int hi) : lo(lo), hi(hi) {}
  Node(vector<int>& v, int lo, int hi) : lo(lo), hi(hi) {
    if (lo + 1 < hi) {
      int mid = lo + (hi - lo) / 2;
      l = new Node(v, lo, mid); r = new Node(v, mid, hi);
      val = max(l->val, r->val);
    } else val = v[lo];
  }
  int query(int L, int R) {
    if (R <= lo || hi <= L) return -inf;
    if (L <= lo && hi <= R) return val;
    push();
    return max(l->query(L, R), r->query(L, R));
  }
  void set(int L, int R, int x) {
    if (R <= lo || hi <= L) return;
    if (L <= lo && hi <= R) mset = val = x, madd = 0;
    else {
      push(), l->set(L, R, x), r->set(L, R, x);
      val = max(l->val, r->val);
    }
  }
  void add(int L, int R, int x) {
    if (R <= lo || hi <= L) return;
    if (L <= lo && hi <= R) {
      if (mset != inf) mset += x;
      else madd += x;
      val += x;
    } else {
      push(), l->add(L, R, x), r->add(L, R, x);
      val = max(l->val, r->val);
    }
  }
  void push() {
    if (!l) {
      int mid = lo + (hi - lo) / 2;
      l = new Node(lo, mid); r = new Node(mid, hi);
    }
    if (mset != inf)
      l->set(lo, hi, mset), r->set(lo, hi, mset), mset = inf;
    else if (madd)
      l->add(lo, hi, madd), r->add(lo, hi, madd), madd = 0;
  }
};

template<typename T>
struct SparseTable {
  int n, K;
  vector<vector<T>> st;
  SparseTable(const vector<T>& a) {
    n = a.size(); K = __lg(n) + 1;
    st.assign(K, vector<T>(n));
    st[0] = a;
    for (int j = 1; j < K; ++j)
      for (int i = 0; i + (1 << j) <= n; ++i)
        st[j][i] = min(st[j - 1][i], st[j - 1][i + (1 << (j - 1))]);
  }
  T query(int L, int R) { // 0-based inclusive range [L, R]
    int j = __lg(R - L + 1);
    return min(st[j][L], st[j][R - (1 << j) + 1]);
  }
};

vector<int> buildLPS(const string &pat) {
  int m = pat.size(), len = 0;
  vector<int> lps(m, 0);
  for (int i = 1; i < m;) {
    if (pat[i] == pat[len]) { len++; lps[i] = len; i++; }
    else {
      if (len != 0) len = lps[len - 1];
      else { lps[i] = 0; i++; }
    }
  }
  return lps;
}
vector<int> KMPsearch(const string &text, const string &pat) {

  vector<int> lps = buildLPS(pat), positions;
  int i = 0, j = 0;
  while (i < n) {
    if (text[i] == pat[j]) {
      i++; j++;
      if (j == m) { positions.push_back(i - j); j = lps[j - 1]; }
    } else {
      if (j != 0) j = lps[j - 1];
      else i++;
    }
  }
  return positions;
}

string get_lcs(const string &s, const string &t) {
  int n = s.size(), m = t.size();
  vector<vector<int>> dp(n + 1, vector<int>(m + 1, 0));
  for (int i = 1; i <= n; i++)
    for (int j = 1; j <= m; j++)
      dp[i][j] = (s[i-1] == t[j-1]) ? 1 + dp[i-1][j-1] : max(dp[i-1][j], dp[i][j-1]);
  string res = ""; int i = n, j = m;
  while (i > 0 && j > 0) {
    if (s[i-1] == t[j-1]) { res += s[i-1]; i--; j--; }
    else if (dp[i-1][j] > dp[i][j-1]) i--;
    else j--;
  }
  reverse(res.begin(), res.end()); return res; // Length = dp[n][m]
}

int edit_distance(const string &s1, const string &s2) {
  int n = s1.size(), m = s2.size();
  vector<vector<int>> dp(n + 1, vector<int>(m + 1));
  for (int i = 0; i <= n; i++) dp[i][0] = i;
  for (int j = 0; j <= m; j++) dp[0][j] = j;
  for (int i = 1; i <= n; i++) {
    for (int j = 1; j <= m; j++) {
      if (s1[i-1] == s2[j-1]) dp[i][j] = dp[i-1][j-1];
      else dp[i][j] = 1 + min({dp[i-1][j], dp[i][j-1], dp[i-1][j-1]});
    }
  }
  return dp[n][m];
}

#include <bits/stdc++.h>
using namespace std;

class DisjointSet {
  vector<int> parent, size;
public:
  DisjointSet(int n) {
    parent.resize(n + 1);
    size.resize(n + 1, 1);
    for (int i = 0; i <= n; i++) parent[i] = i;
  }
  int findUPar(int node) {
    if (node == parent[node]) return node;
    return parent[node] = findUPar(parent[node]);
  }
  void unionBySize(int u, int v) {
    int ulpU = findUPar(u), ulpV = findUPar(v);
    if (ulpU == ulpV) return;
    if (size[ulpU] < size[ulpV]) {
      parent[ulpU] = ulpV; size[ulpV] += size[ulpU];
    } else {
      parent[ulpV] = ulpU; size[ulpU] += size[ulpV];
    }
  }
  bool connected(int u, int v) { return findUPar(u) == findUPar(v); }
};

struct Edge {
  int u, v, w;
  bool operator<(const Edge& o) const { return w < o.w; }
};

int kruskal(int n, vector<Edge>& edges) {
  sort(edges.begin(), edges.end());
  DisjointSet dsu(n);
  int cost = 0, count = 0;
  for (auto& e : edges) {
    if (!dsu.connected(e.u, e.v)) {
      dsu.unionBySize(e.u, e.v);
      cost += e.w;
      count++;
      if (count == n - 1) return cost;
    }
  }
  return -1;
}

int main() {
  ios::sync_with_stdio(false);
  cin.tie(nullptr);
  int n, m; cin >> n >> m;
  vector<Edge> edges(m);
  for (int i = 0; i < m; i++) cin >> edges[i].u >> edges[i].v >> edges[i].w;
  int ans = kruskal(n, edges);
  if (ans == -1) cout << "MST does not exist\n";
  else cout << "Minimum cost = " << ans << '\n';
  return 0;
}
