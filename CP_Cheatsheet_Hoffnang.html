Count Set Bits (1s) from 0 to N numofonef | O(log N) 
int numofone(int n) {
  int ans = 0;
  for (int i = 0; (1ULL << i) <= n; i++) {
    int vag = 1ULL << (i + 1), arektavag = (n + 1) / vag;
    ans += (arektavag * (vag >> 1ULL));
    int obos = (n + 1) % vag;
    if (obos > (vag >> 1ULL)) ans += ((obos) - (vag >> 1ULL));
    if (arektavag == 0) break;
  }
  return ans;
}

Combinatorics: nCr in O(R) 
int nCr(int n, int r) {
  if (r < 0 || r > n) return 0;
  if (r > n / 2) r = n - r;
  int ans = 1;
  for (int i = 1; i <= r; ++i) ans = ans * (n - i + 1) / i;
  return ans;
}
Stable Quadratic Equation Solver quadf | O(1)
Citardauq formula avoids catastrophic cancellation when b ≈ √D. Handles a=0 linear, D<0, D=0.
using ld = long double;
const ld EPS = 1e-12;

void solve_quadratic(ld a, ld b, ld c) {
  cout << fixed << setprecision(12);
  if (fabsl(a) < EPS) {
    if (fabsl(b) < EPS) cout << "No solution\n";
    else cout << -c / b << '\n';
    return;
  }
  ld d = b * b - 4 * a * c;
  if (d < -EPS) { cout << "No real roots\n"; return; }
  if (fabsl(d) <= EPS) { cout << -b / (2 * a) << '\n'; return; }
  ld sq = sqrtl(d);
  ld q = -0.5L * (b + copysignl(sq, b)); // Numerically stable
  ld x1 = q / a, x2 = c / q;
  if (x1 > x2) swap(x1, x2);
  cout << x1 << '\n' << x2 << '\n';
}

Topological Sort (Kahn's BFS) O(V + E) 
vector<int> topo_sort(int n, const vector<vector<int>> &adj) {
  vector<int> in_deg(n, 0), order;
  for (int u = 0; u < n; ++u)
    for (int v : adj[u]) in_deg[v]++;
  queue<int> q;
  for (int i = 0; i < n; ++i) if (in_deg[i] == 0) q.push(i);
  while (!q.empty()) {
    int u = q.front(); q.pop(); order.push_back(u);
    for (int v : adj[u]) if (--in_deg[v] == 0) q.push(v);
  }
  return (order.size() == n) ? order : vector<int>();
}

const int INF = 1000000000;
vector<vector<pair<int, int>>> adj;
void dijkstra(int s, vector<int> & d, vector<int> & p) {
  int n = adj.size(); d.assign(n, INF); p.assign(n, -1);
  d[s] = 0; set<pair<int, int>> q; q.insert({0, s});
  while (!q.empty()) {
    int v = q.begin()->second; q.erase(q.begin());
    for (auto edge : adj[v]) {
      int to = edge.first, len = edge.second;
      if (d[v] + len < d[to]) {
        q.erase({d[to], to}); d[to] = d[v] + len; p[to] = v;
        q.insert({d[to], to});
      }
    }
  }
}
Bellman-Ford Algorithm bellman_ford | O(V * E) 
vector<int> bellman_ford(int n, vector<vector<int>>& edges, int S) {
  vector<int> dis(n, 1e9); dis[S] = 0;
  for (int i = 0; i < n - 1; i++) {
    bool updated = false;
    for (auto &e : edges) {
      int u = e[0], v = e[1], w = e[2];
      if (dis[u] != 1e9 && dis[u] + w < dis[v]) {
        dis[v] = dis[u] + w; updated = true;
      }
    }
    if (!updated) break;
  }
  for (auto &e : edges) {
    int u = e[0], v = e[1], w = e[2];
    if (dis[u] != 1e9 && dis[u] + w < dis[v]) return {-1};
  }
  return dis;
}

pair<int, int> bfs_farthest(int start, const vector<vector<int>>& adj, int n) {
  vector<int> dist(n + 1, -1);
  queue<int> q; q.push(start); dist[start] = 0;
  int farthest_node = start;
  while (!q.empty()) {
    int node = q.front(); q.pop();
    for (int neighbor : adj[node]) {
      if (dist[neighbor] == -1) {
        dist[neighbor] = dist[node] + 1; q.push(neighbor);
        if (dist[neighbor] > dist[farthest_node]) farthest_node = neighbor;
      }
    }
  }
  return {farthest_node, dist[farthest_node]};
}

7. Data Structures (DSU & Range Trees)
Disjoint Set Union (Union by Size) dsu-size | O(α(N)) ≈ O(1) 
class DisjointSet {
  vector<int> rank, parent, size;
public:
  DisjointSet(int n) {
    rank.resize(n + 1, 0); parent.resize(n + 1); size.resize(n + 1, 1);
    for (int i = 0; i <= n; i++) parent[i] = i;
  }
  int findUPar(int node) {
    if (node == parent[node]) return node;
    return parent[node] = findUPar(parent[node]);
  }
  void unionBySize(int u, int v) {
    int ulpU = findUPar(u), ulpV = findUPar(v);
    if (ulpU == ulpV) return;
    if (size[ulpU] < size[ulpV]) {
      parent[ulpU] = ulpV; size[ulpV] += size[ulpU];
    } else {
      parent[ulpV] = ulpU; size[ulpU] += size[ulpV];
    }
  }
  bool connected(int u, int v) { return findUPar(u) == findUPar(v); }
};
Fenwick Tree (BIT - Prefix Sums) O(log N) 
struct Fenwick {
  int n; vector<int> bit;
  Fenwick(int n) : n(n), bit(n + 1, 0) {}
  void add(int idx, int val) {
    for (; idx <= n; idx += idx & -idx) bit[idx] += val;
  }
  int query(int idx) {
    int sum = 0;
    for (; idx > 0; idx -= idx & -idx) sum += bit[idx];
    return sum;
  }
  int range_query(int l, int r) { return query(r) - query(l - 1); }
};
Iterative Segment Tree (Range Sum) O(log N) 
struct SegTree {
  int n; vector<int> t;
  SegTree(int n) : n(n), t(2 * n, 0) {}
  void update(int p, int val) {
    for (t[p += n] = val; p > 1; p >>= 1) t[p >> 1] = t[p] + t[p ^ 1];
  }
  int query(int l, int r) { // sum on [l, r]
    int res = 0;
    for (l += n, r += n + 1; l < r; l >>= 1, r >>= 1) {
      if (l & 1) res += t[l++];
      if (r & 1) res += t[--r];
    }
    return res;
  }
};
8. String Algorithms (KMP)
Knuth-Morris-Pratt (KMP) Algorithm kmp | O(N + M) 
vector<int> buildLPS(const string &pat) {
  int m = pat.size(), len = 0;
  vector<int> lps(m, 0);
  for (int i = 1; i < m;) {
    if (pat[i] == pat[len]) { len++; lps[i] = len; i++; }
    else {
      if (len != 0) len = lps[len - 1];
      else { lps[i] = 0; i++; }
    }
  }
  return lps;
}
vector<int> KMPsearch(const string &text, const string &pat) {
  int n = text.size(), m = pat.size();
  vector<int> lps = buildLPS(pat), positions;
  int i = 0, j = 0;
  while (i < n) {
    if (text[i] == pat[j]) {
      i++; j++;
      if (j == m) { positions.push_back(i - j); j = lps[j - 1]; }
    } else {
      if (j != 0) j = lps[j - 1];
      else i++;
    }
  }
  return positions;
}

Longest Increasing Subsequence (LIS) O(N log N) 
int lis(const vector<int>& a) {
  vector<int> tails;
  for (int x : a) {
    auto it = lower_bound(tails.begin(), tails.end(), x);
    if (it == tails.end()) tails.push_back(x);
    else *it = x;
  }
  return tails.size();
}

Kruskal's Minimum Spanning Tree (MST) O(E log E) 
struct Edge {
  int u, v, w;
  bool operator<(const Edge& o) const { return w < o.w; }
};
int kruskal(int n, vector<Edge>& edges) {
  sort(all(edges));
  DisjointSet dsu(n);
  int cost = 0, count = 0;
  for (auto& e : edges) {
    if (!dsu.connected(e.u, e.v)) {
      dsu.unionBySize(e.u, e.v); cost += e.w;
      if (++count == n - 1) return cost;
    }
  }
  return -1;
}
Matrix Exponentiation (O(K^3 log N)) 
template<typename T = long long>
struct Matrix {
  int n; vector<vector<T>> a;
  Matrix(int n) : n(n), a(n, vector<T>(n, 0)) {}
  static Matrix eye(int n) {
    Matrix res(n); for (int i = 0; i < n; i++) res.a[i][i] = 1; return res;
  }
  Matrix operator*(const Matrix& o) const {
    Matrix res(n);
    for (int i = 0; i < n; i++)
      for (int k = 0; k < n; k++)
        for (int j = 0; j < n; j++)
          res.a[i][j] = (res.a[i][j] + a[i][k] * o.a[k][j]) % 1000000007;
    return res;
  }
  static Matrix power(Matrix m, long long p) {
    Matrix res = Matrix::eye(m.n);
    while (p) {
      if (p & 1) res = res * m;
      m = m * m; p >>= 1;
    }
    return res;
  }
};
Shoelace Formula (Polygon Area) O(N) 
struct Point { double x, y; };
double polygon_area(const vector<Point>& p) {
  double area = 0.0; int n = p.size();
  for (int i = 0; i < n; ++i) {
    int j = (i + 1) % n;
    area += p[i].x * p[j].y - p[j].x * p[i].y;
  }
  return fabs(area) / 2.0;
